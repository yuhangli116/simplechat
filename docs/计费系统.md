## 二、钻石消耗计算

### 2.1 计算公式

```
消耗钻石 = 输入Token × 输入倍率
         + 输出Token × 输出倍率
         + 思考Token × 思考倍率
         + 缓存Token × 缓存倍率

结果向上取整
```

### 2.2 计算示例

**示例1：使用 DeepSeek-V3 写小说**

```
输入：3000 Token（约4500字背景材料）
输出：3000 Token（约4500字正文）

计算：
- 输入消耗：3000 × 1 = 3,000 钻石
- 输出消耗：3000 × 4 = 12,000 钻石
- 总消耗：15,000 钻石

折合人民币：15,000 ÷ 250,000 = 0.06元
```

**示例2：使用 Claude Sonnet 写小说**

```
输入：3000 Token
输出：3000 Token

计算：
- 输入消耗：3000 × 10.5 = 31,500 钻石
- 输出消耗：3000 × 52.5 = 157,500 钻石
- 总消耗：189,000 钻石

折合人民币：189,000 ÷ 250,000 = 0.76元
```

**示例3：使用 DeepSeek-R1 深度推理**

```
输入：5000 Token
输出：2000 Token
思考：8000 Token

计算：
- 输入消耗：5000 × 2 = 10,000 钻石
- 输出消耗：2000 × 8 = 16,000 钻石

折合人民币：90,000 ÷ 250,000 = 0.36元
```

## 八、核心业务逻辑

### 8.1 钻石扣减存储过程

```sql
CREATE OR REPLACE FUNCTION deduct_diamonds(
    p_user_id UUID,
    p_model_key VARCHAR,
    p_input_tokens INTEGER,
    p_output_tokens INTEGER,
    p_reasoning_tokens INTEGER DEFAULT 0,
    p_cache_tokens INTEGER DEFAULT 0
) RETURNS JSON AS $$
DECLARE
    v_diamonds_needed INTEGER;
    v_member_diamonds INTEGER;
    v_permanent_diamonds INTEGER;
    v_member_to_use INTEGER := 0;
    v_permanent_to_use INTEGER := 0;
    v_input_mult DECIMAL;
    v_output_mult DECIMAL;
    v_reasoning_mult DECIMAL;
    v_cache_mult DECIMAL;
    v_estimated_cost DECIMAL(10, 6);
BEGIN
    -- 1. 获取模型倍率
    SELECT input_multiplier, output_multiplier, reasoning_multiplier, cache_multiplier
    INTO v_input_mult, v_output_mult, v_reasoning_mult, v_cache_mult
    FROM model_pricing
    WHERE model_key = p_model_key AND is_active = TRUE;

    IF NOT FOUND THEN
        RETURN json_build_object('success', FALSE, 'error', '模型不存在');
    END IF;

    -- 2. 计算所需钻石
    v_diamonds_needed := CEIL(
        p_input_tokens * v_input_mult +
        p_output_tokens * v_output_mult +
        p_reasoning_tokens * v_reasoning_mult +
        p_cache_tokens * v_cache_mult
    );

    -- 3. 获取用户余额（加锁）
    SELECT member_diamonds, permanent_diamonds
    INTO v_member_diamonds, v_permanent_diamonds
    FROM profiles
    WHERE id = p_user_id
    FOR UPDATE;

    IF NOT FOUND THEN
        RETURN json_build_object('success', FALSE, 'error', '用户不存在');
    END IF;

    -- 4. 检查余额是否充足
    IF v_member_diamonds + v_permanent_diamonds < v_diamonds_needed THEN
        RETURN json_build_object(
            'success', FALSE,
            'error', '钻石不足',
            'needed', v_diamonds_needed,
            'available', v_member_diamonds + v_permanent_diamonds
        );
    END IF;

    -- 5. 计算扣减方案（优先扣会员钻石）
    IF v_member_diamonds >= v_diamonds_needed THEN
        v_member_to_use := v_diamonds_needed;
        v_permanent_to_use := 0;
    ELSE
        v_member_to_use := v_member_diamonds;
        v_permanent_to_use := v_diamonds_needed - v_member_diamonds;
    END IF;

    -- 6. 执行扣减
    UPDATE profiles
    SET member_diamonds = member_diamonds - v_member_to_use,
        permanent_diamonds = permanent_diamonds - v_permanent_to_use,
        total_diamonds_consumed = total_diamonds_consumed + v_diamonds_needed,
        updated_at = NOW()
    WHERE id = p_user_id;

    -- 7. 估算成本（用于统计）
    v_estimated_cost := v_diamonds_needed / 250000.0 * 0.5;

    -- 8. 记录消费日志
    INSERT INTO usage_logs (
        user_id, model_name,
        input_tokens, output_tokens, reasoning_tokens, cache_tokens,
        diamonds_consumed, member_diamonds_used, permanent_diamonds_used,
        estimated_cost_cny
    ) VALUES (
        p_user_id, p_model_key,
        p_input_tokens, p_output_tokens, p_reasoning_tokens, p_cache_tokens,
        v_diamonds_needed, v_member_to_use, v_permanent_to_use,
        v_estimated_cost
    );

    -- 9. 返回结果
    RETURN json_build_object(
        'success', TRUE,
        'diamonds_consumed', v_diamonds_needed,
        'member_diamonds_remaining', v_member_diamonds - v_member_to_use,
        'permanent_diamonds_remaining', v_permanent_diamonds - v_permanent_to_use,
        'total_remaining', (v_member_diamonds - v_member_to_use) + (v_permanent_diamonds - v_permanent_to_use)
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### 8.2 会员购买处理

```sql
CREATE OR REPLACE FUNCTION process_membership_purchase(
    p_user_id UUID,
    p_order_id UUID,
    p_membership_type VARCHAR,
    p_diamonds INTEGER
) RETURNS JSON AS $$
DECLARE
    v_current_type VARCHAR;
    v_current_expires TIMESTAMP;
    v_new_expires TIMESTAMP;
    v_days INTEGER;
BEGIN
    -- 1. 锁定订单（幂等性保证）
    UPDATE recharge_logs
    SET status = 'success', paid_at = NOW()
    WHERE id = p_order_id AND status = 'pending';

    IF NOT FOUND THEN
        RETURN json_build_object('success', FALSE, 'error', '订单不存在或已处理');
    END IF;

    -- 2. 确定会员天数
    v_days := CASE p_membership_type
        WHEN 'monthly' THEN 30
        WHEN 'quarterly' THEN 90
        WHEN 'yearly' THEN 365
        ELSE 0
    END;

    -- 3. 获取当前会员状态
    SELECT membership_type, membership_expires_at
    INTO v_current_type, v_current_expires
    FROM profiles WHERE id = p_user_id;

    -- 4. 计算新的过期时间
    IF v_current_expires IS NULL OR v_current_expires < NOW() THEN
        v_new_expires := NOW() + (v_days || ' days')::INTERVAL;
    ELSE
        v_new_expires := v_current_expires + (v_days || ' days')::INTERVAL;
    END IF;

    -- 5. 更新用户信息
    UPDATE profiles
    SET membership_type = p_membership_type,
        membership_expires_at = v_new_expires,
        member_diamonds = member_diamonds + p_diamonds,
        updated_at = NOW()
    WHERE id = p_user_id;

    -- 6. 记录发放日志
    INSERT INTO member_diamond_logs (user_id, diamonds, membership_type, expires_at)
    VALUES (p_user_id, p_diamonds, p_membership_type, v_new_expires);

    RETURN json_build_object(
        'success', TRUE,
        'membership_type', p_membership_type,
        'expires_at', v_new_expires,
        'diamonds_added', p_diamonds
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```typescript
/**
 * 计算钻石消耗
 */
export function calculateDiamonds(
  modelKey: string,
  inputTokens: number,
  outputTokens: number,
  reasoningTokens: number = 0,
  cacheTokens: number = 0
): number {
  const pricing = MODEL_PRICING[modelKey];
  if (!pricing) return 0;

  return Math.ceil(
    inputTokens * pricing.inputMultiplier +
    outputTokens * pricing.outputMultiplier +
    reasoningTokens * pricing.reasoningMultiplier +
    cacheTokens * pricing.cacheMultiplier
  );
}

/**
 * 格式化钻石数量显示
 */
export function formatDiamonds(diamonds: number): string {
  if (diamonds >= 100000000) {
    return `${(diamonds / 100000000).toFixed(2)}亿`;
  }
  if (diamonds >= 10000) {
    return `${(diamonds / 10000).toFixed(1)}万`;
  }
  return diamonds.toLocaleString();
}

/**
 * 钻石转人民币显示
 */
export function diamondsToYuan(diamonds: number): string {
  const yuan = diamonds / PRICING_CONFIG.DIAMONDS_PER_YUAN;
  if (yuan < 0.01) return '不到1分';
  if (yuan < 1) return `约${Math.ceil(yuan * 100)}分`;
  return `约¥${yuan.toFixed(2)}`;
}

/**
 * 调用扣费接口
 */
export async function deductDiamonds(
  userId: string,
  modelKey: string,
  inputTokens: number,
  outputTokens: number,
  reasoningTokens: number = 0,
  cacheTokens: number = 0
): Promise<{
  success: boolean;
  error?: string;
  diamondsConsumed?: number;
  totalRemaining?: number;
}> {
  const { data, error } = await supabase.rpc('deduct_diamonds', {
    p_user_id: userId,
    p_model_key: modelKey,
    p_input_tokens: inputTokens,
    p_output_tokens: outputTokens,
    p_reasoning_tokens: reasoningTokens,
    p_cache_tokens: cacheTokens,
  });

  if (error) {
    return { success: false, error: error.message };
  }

  return {
    success: data.success,
    error: data.error,
    diamondsConsumed: data.diamonds_consumed,
    totalRemaining: data.total_remaining,
  };
}

/**
 * 获取用户余额
 */
export async function getUserBalance(userId: string): Promise<{
  memberDiamonds: number;
  permanentDiamonds: number;
  totalDiamonds: number;
}> {
  const { data, error } = await supabase
    .from('profiles')
    .select('member_diamonds, permanent_diamonds')
    .eq('id', userId)
    .single();

  if (error || !data) {
    return { memberDiamonds: 0, permanentDiamonds: 0, totalDiamonds: 0 };
  }

  return {
    memberDiamonds: data.member_diamonds || 0,
    permanentDiamonds: data.permanent_diamonds || 0,
    totalDiamonds: (data.member_diamonds || 0) + (data.permanent_diamonds || 0),
  };
}
```
